# V1.3: FTS 全文搜索 + SHA256 异步化 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add SQLite FTS5 full-text content search across Tier 1-2 file types (txt/md/log/csv/pdf/docx/xlsx) and offload SHA256 hashing to background threads to eliminate UI blocking.

**Architecture:** TextExtractor service (format dispatch + encoding detection + timeout) → IndexWorker (background QObject queue) → FTS5 virtual table; ImportWorker (background QThread) for async SHA256 + DB upsert; `search_files()` expands to FTS primary + LIKE fallback.

**Tech Stack:** Python 3.11, PyQt5, SQLite FTS5, pymupdf, python-docx, openpyxl, chardet

## Global Constraints

- FTS5 content field stores `f"{file_name} {extracted_text}"` for unified file-name + content search
- MAX_TEXT_LENGTH = 100_000 chars per file
- MAX_FILE_SIZE = 50 MB per file (skip larger)
- EXTRACT_TIMEOUT = 5.0 seconds per file
- chardet detection limited to first 10 KB of file
- IndexWorker queue capacity: 10,000 items (drop oldest on overflow)
- Each 50 MB / 100 KB / 5 s limit enforced with test
- All new DB operations use their own connection in background threads (never share connection across threads)
- `file_content_status.status` transitions must be immediate (no zombie `pending`)
- Progressive indexing batch size: 100 files per batch, 200 ms interval between batches
- Sorting: FTS results first (BM25 rank), LIKE supplements appended (file_name ASC)
- PDF scan detection: if PyMuPDF returns < 50 chars of text total, flag as "no searchable text"
- Docx/xlsx extraction: do NOT handle tracked changes/comments/formulas — pure text only

```python
# src/app/services/text_extractor.py
import logging
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor, TimeoutError

import chardet

logger = logging.getLogger(__name__)

MAX_FILE_SIZE = 50 * 1024 * 1024       # 50 MB
MAX_TEXT_LENGTH = 100_000              # 100K chars
EXTRACT_TIMEOUT = 5.0                  # seconds
CHARDET_SAMPLE_SIZE = 10 * 1024        # 10 KB
DEFAULT_ENCODINGS = ["utf-8", "utf-16", "gbk", "gb2312", "latin-1"]

EXTENSIONS_TXT = {".txt", ".md", ".log", ".csv"}
EXTENSIONS_PDF = {".pdf"}
EXTENSIONS_DOCX = {".docx"}
EXTENSIONS_XLSX = {".xlsx"}


def extract_text(path: Path, file_ext: str) -> str:
    ext = f".{file_ext.lstrip('.').lower()}"
    if ext in EXTENSIONS_TXT:
        return extract_txt(path)
    if ext in EXTENSIONS_PDF:
        return extract_pdf(path)
    if ext in EXTENSIONS_DOCX:
        return extract_docx(path)
    if ext in EXTENSIONS_XLSX:
        return extract_xlsx(path)
    return ""


def _run_with_timeout(fn, path: Path, timeout: float = EXTRACT_TIMEOUT):
    with ThreadPoolExecutor(max_workers=1) as pool:
        future = pool.submit(fn, path)
        try:
            return future.result(timeout=timeout)
        except TimeoutError:
            logger.warning("Extraction timed out: %s", path)
            return ""
        except Exception:
            logger.exception("Extraction failed: %s", path)
            return ""


def extract_txt(path: Path) -> str:
    if path.stat().st_size > MAX_FILE_SIZE:
        return ""
    sample = _read_sample(path)
    encoding = _detect_encoding(sample)
    try:
        text = path.read_text(encoding=encoding, errors="replace")
    except (LookupError, UnicodeDecodeError):
        for enc in DEFAULT_ENCODINGS:
            try:
                text = path.read_text(encoding=enc, errors="replace")
                break
            except (UnicodeDecodeError, LookupError):
                continue
        else:
            return ""
    return text[:MAX_TEXT_LENGTH]


def _read_sample(path: Path) -> bytes:
    with path.open("rb") as f:
        return f.read(CHARDET_SAMPLE_SIZE)


def _detect_encoding(sample: bytes) -> str:
    result = chardet.detect(sample)
    return result.get("encoding", "utf-8") or "utf-8"


def extract_pdf(path: Path) -> str:
    import fitz
    if path.stat().st_size > MAX_FILE_SIZE:
        return ""
    text_pages = []
    try:
        doc = fitz.open(str(path))
        for page in doc:
            text_pages.append(page.get_text())
        doc.close()
    except Exception:
        logger.exception("PDF extraction failed: %s", path)
        return ""
    full = " ".join(text_pages)
    if len(full.strip()) < 50:
        return ""
    return full[:MAX_TEXT_LENGTH]


def extract_docx(path: Path) -> str:
    from docx import Document
    if path.stat().st_size > MAX_FILE_SIZE:
        return ""
    try:
        doc = Document(str(path))
        parts = [p.text for p in doc.paragraphs]
        for table in doc.tables:
            for row in table.rows:
                for cell in row.cells:
                    parts.append(cell.text)
        return " ".join(parts)[:MAX_TEXT_LENGTH]
    except Exception:
        logger.exception("DOCX extraction failed: %s", path)
        return ""


def extract_xlsx(path: Path) -> str:
    from openpyxl import load_workbook
    if path.stat().st_size > MAX_FILE_SIZE:
        return ""
    try:
        wb = load_workbook(str(path), read_only=True, data_only=True)
        parts = []
        for sheet in wb.worksheets:
            for row in sheet.iter_rows():
                for cell in row:
                    if cell.value is not None:
                        parts.append(str(cell.value))
        wb.close()
        return " ".join(parts)[:MAX_TEXT_LENGTH]
    except Exception:
        logger.exception("XLSX extraction failed: %s", path)
        return ""
```

```python
def create_fts_table(self, conn: Optional[sqlite3.Connection] = None) -> None:
    own_conn = conn is None
    if own_conn:
        conn = self.connect()
    try:
        conn.execute("""
            CREATE VIRTUAL TABLE IF NOT EXISTS file_fts USING fts5(
                file_id UNINDEXED,
                content,
                tokenize='unicode61'
            )
        """)
        conn.execute("""
            CREATE TABLE IF NOT EXISTS file_content_status (
                file_id INTEGER PRIMARY KEY,
                content_length INTEGER NOT NULL DEFAULT 0,
                status TEXT NOT NULL DEFAULT 'pending',
                error_message TEXT,
                indexed_at INTEGER,
                FOREIGN KEY (file_id) REFERENCES files(id) ON DELETE CASCADE
            )
        """)
    finally:
        if own_conn:
            conn.close()

def upsert_fts(self, conn: sqlite3.Connection, file_id: int, file_name: str, content: str) -> None:
    fts_content = f"{file_name} {content}"
    conn.execute(
        "INSERT INTO file_fts (file_id, content) VALUES (?, ?) "
        "ON CONFLICT(file_id) DO UPDATE SET content = excluded.content",
        (file_id, fts_content),
    )
    conn.execute(
        """INSERT INTO file_content_status (file_id, content_length, status, indexed_at)
           VALUES (?, ?, 'indexed', ?)
           ON CONFLICT(file_id) DO UPDATE SET
               content_length = excluded.content_length,
               status = 'indexed',
               error_message = NULL,
               indexed_at = excluded.indexed_at""",
        (file_id, len(fts_content), int(time.time())),
    )

def delete_fts(self, conn: sqlite3.Connection, file_id: int) -> None:
    conn.execute("DELETE FROM file_fts WHERE file_id = ?", (file_id,))
    conn.execute("DELETE FROM file_content_status WHERE file_id = ?", (file_id,))

def search_fts(self, conn: sqlite3.Connection, query: str, limit: int = 200) -> list[dict]:
    rows = conn.execute(
        """SELECT f.id AS file_id, f.file_name, f.full_path, f.file_ext,
                  f.file_size, f.modify_time, rank
           FROM file_fts fts
           JOIN files f ON f.id = fts.file_id
           WHERE f.is_deleted = 0 AND file_fts MATCH ?
           ORDER BY rank
           LIMIT ?""",
        (query, limit),
    ).fetchall()
    return [dict(r) for r in rows]

def get_file_content_status(self, conn: sqlite3.Connection, file_id: int) -> Optional[dict]:
    row = conn.execute(
        "SELECT * FROM file_content_status WHERE file_id = ?", (file_id,)
    ).fetchone()
    return dict(row) if row else None
```

```python
# src/app/services/index_worker.py
import logging
import time
from pathlib import Path
from queue import Queue, Full as QueueFull
from typing import Optional

from PyQt5.QtCore import QObject, QThread, pyqtSignal

from src.app.db.database import Database
from src.app.services.text_extractor import extract_text

logger = logging.getLogger(__name__)

INDEX_BATCH_SIZE = 100
INDEX_BATCH_INTERVAL_MS = 200
QUEUE_MAXSIZE = 10000

SUPPORTED_EXTENSIONS = {".txt", ".md", ".log", ".csv", ".pdf", ".docx", ".xlsx"}


class IndexWorker(QObject):
    indexing_progress = pyqtSignal(int, int)   # current, total
    indexing_finished = pyqtSignal()
    index_updated = pyqtSignal(int)            # file_id

    def __init__(self, db: Database, parent: Optional[QObject] = None):
        super().__init__(parent)
        self._db = db
        self._queue: Queue = Queue(maxsize=QUEUE_MAXSIZE)
        self._total = 0
        self._processed = 0
        self._running = False
        self._thread: Optional[QThread] = None

    def enqueue(self, file_id: int, path: Path) -> None:
        ext = path.suffix.lower()
        if ext not in SUPPORTED_EXTENSIONS:
            return
        try:
            self._queue.put_nowait((file_id, path))
        except QueueFull:
            logger.warning("Index queue full, dropping oldest")
            try:
                self._queue.get_nowait()
                self._queue.put_nowait((file_id, path))
            except QueueFull:
                pass

    def enqueue_batch(self, items: list[tuple[int, Path]]) -> None:
        for file_id, path in items:
            self.enqueue(file_id, path)
        self._total = self._queue.qsize()
        self._processed = 0

    def start(self) -> None:
        if self._running:
            return
        self._running = True
        self._thread = QThread()
        self.moveToThread(self._thread)
        self._thread.started.connect(self._process_queue)
        self._thread.start()

    def stop(self) -> None:
        self._running = False
        if self._thread:
            self._thread.quit()
            self._thread.wait(2000)
            self._thread = None

    def _process_queue(self) -> None:
        conn = self._db.connect()
        self._db.create_fts_table(conn)
        while self._running:
            processed_in_batch = 0
            while processed_in_batch < INDEX_BATCH_SIZE and self._running:
                try:
                    file_id, path = self._queue.get(timeout=0.1)
                except Exception:
                    continue
                self._index_one(conn, file_id, path)
                processed_in_batch += 1
                self._processed += 1
                self.indexing_progress.emit(self._processed, self._total)
            if processed_in_batch > 0:
                conn.commit()
                QThread.msleep(INDEX_BATCH_INTERVAL_MS)
            if self._queue.empty() and self._processed >= self._total:
                self.indexing_finished.emit()
                break
        conn.close()

    def _index_one(self, conn, file_id: int, path: Path) -> None:
        try:
            ext = path.suffix.lower().lstrip(".")
            text = extract_text(path, ext)
            if text:
                file_name = path.name
                self._db.upsert_fts(conn, file_id, file_name, text)
            else:
                reason = "skipped_too_large" if path.stat().st_size > 50*1024*1024 else "skipped_format"
                conn.execute(
                    "INSERT INTO file_content_status (file_id, content_length, status, indexed_at) "
                    "VALUES (?, 0, ?, ?) ON CONFLICT(file_id) DO UPDATE SET "
                    "status = excluded.status, error_message = NULL, indexed_at = excluded.indexed_at",
                    (file_id, reason, int(time.time())),
                )
            self.index_updated.emit(file_id)
        except Exception:
            logger.exception("Index failed for file_id=%s path=%s", file_id, path)
            conn.execute(
                "INSERT INTO file_content_status (file_id, content_length, status, error_message, indexed_at) "
                "VALUES (?, 0, 'error', ?, ?) ON CONFLICT(file_id) DO UPDATE SET "
                "status = 'error', error_message = excluded.error_message, indexed_at = excluded.indexed_at",
                (file_id, str(e), int(time.time())),
            )
            self.index_updated.emit(file_id)
```

```python
import hashlib
import logging
import time
from pathlib import Path
from typing import Optional

from PyQt5.QtCore import QObject, QThread, pyqtSignal

from src.app.db.database import Database
from src.app.viewmodel.main_viewmodel import ImportCandidate, sha256_file

logger = logging.getLogger(__name__)


class ImportWorker(QObject):
    progress = pyqtSignal(int, int)   # current, total
    finished = pyqtSignal()
    file_imported = pyqtSignal(int, str)  # file_id, file_path

    def __init__(self, db: Database, candidates: list[ImportCandidate], parent: Optional[QObject] = None):
        super().__init__(parent)
        self._db = db
        self._candidates = candidates
        self._thread: Optional[QThread] = None

    def run(self):
        self._thread = QThread()
        self.moveToThread(self._thread)
        self._thread.started.connect(self._do_import)
        self._thread.start()

    def _do_import(self):
        now = int(time.time())
        inserted = 0
        conn = self._db.connect()
        total = len(self._candidates)
        for i, c in enumerate(self._candidates):
            try:
                file_hash = sha256_file(c.path)
            except Exception:
                logger.exception("SHA256 failed: %s", c.path)
                self.progress.emit(i + 1, total)
                continue
            file_name = c.path.name
            file_ext = c.path.suffix.lower().lstrip(".") if c.path.suffix else None
            full_path = str(c.path.resolve())
            try:
                st = c.path.stat()
            except OSError:
                self.progress.emit(i + 1, total)
                continue
            row = conn.execute(
                "SELECT id FROM files WHERE file_hash = ?", (file_hash,)
            ).fetchone()
            if row is None:
                conn.execute(
                    """INSERT INTO files (file_name, file_ext, full_path, file_size,
                       create_time, modify_time, file_hash, is_deleted, created_at, updated_at)
                       VALUES (?, ?, ?, ?, ?, ?, ?, 0, ?, ?)""",
                    (file_name, file_ext, full_path, int(st.st_size),
                     int(st.st_ctime), int(st.st_mtime), file_hash, now, now),
                )
                file_id = conn.execute("SELECT last_insert_rowid()").fetchone()[0]
                inserted += 1
            else:
                file_id = row["id"]
                conn.execute(
                    """UPDATE files SET file_name=?, file_ext=?, full_path=?, file_size=?,
                       create_time=?, modify_time=?, is_deleted=0, deleted_at=NULL, updated_at=?
                       WHERE file_hash=?""",
                    (file_name, file_ext, full_path, int(st.st_size),
                     int(st.st_ctime), int(st.st_mtime), now, file_hash),
                )
            self.file_imported.emit(file_id, full_path)
            self.progress.emit(i + 1, total)
        conn.commit()
        conn.close()
        self.finished.emit()
        logger.info("Async import finished: %s new files", inserted)
        if self._thread:
            self._thread.quit()
```
