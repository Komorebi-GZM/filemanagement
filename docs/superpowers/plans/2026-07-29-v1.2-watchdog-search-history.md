```python
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Optional

from PyQt5.QtCore import QObject, QTimer, pyqtSignal
from watchdog.events import FileSystemEvent, PatternMatchingEventHandler
from watchdog.observers import Observer


@dataclass
class FileEvent:
    event_type: str  # "created" | "modified" | "deleted" | "moved"
    src_path: str
    dest_path: str  # only for "moved"
    file_hash: str  # empty string if unknown
    is_directory: bool


class _FileChangeHandler(PatternMatchingEventHandler):
    IGNORE_PATTERNS = [
        "*.tmp", "*.temp", "~$*", "*.swp", "*.swx",
        "*.lnk", "Thumbs.db", ".DS_Store",
    ]
    IGNORE_DIRS = [".git", "__pycache__", ".svn"]

    def __init__(self, queue_fn):
        super().__init__(
            patterns=["*"],
            ignore_patterns=self.IGNORE_PATTERNS,
            ignore_directories=True,
            case_sensitive=False,
        )
        self._queue_fn = queue_fn

    def on_modified(self, event: FileSystemEvent):
        self._queue_fn("modified", src_path=event.src_path)

    def on_created(self, event: FileSystemEvent):
        self._queue_fn("created", src_path=event.src_path)

    def on_deleted(self, event: FileSystemEvent):
        self._queue_fn("deleted", src_path=event.src_path)

    def on_moved(self, event: FileSystemEvent):
        self._queue_fn(
            "moved",
            src_path=event.src_path,
            dest_path=event.dest_path,
        )


class FileMonitor(QObject):
    file_system_changed = pyqtSignal(object)  # list[FileEvent]

    def __init__(self, debounce_ms: int = 2000, parent: Optional[QObject] = None):
        super().__init__(parent)
        self._debounce_ms = debounce_ms
        self._pending: dict[str, FileEvent] = {}
        self._observer = Observer()
        self._handlers: list[_FileChangeHandler] = []
        self._watch_dirs: list[Path] = []

        self._timer = QTimer(self)
        self._timer.setSingleShot(True)
        self._timer.timeout.connect(self._flush)

    def start(self, watch_dirs: list[Path]) -> None:
        self._watch_dirs = watch_dirs
        for d in watch_dirs:
            if not d.exists():
                continue
            h = _FileChangeHandler(self._queue_event)
            self._handlers.append(h)
            self._observer.schedule(h, str(d), recursive=True)

    def run(self) -> None:
        """Start the observer thread. Call after start()."""
        self._observer.start()

    def stop(self) -> None:
        self._timer.stop()
        if self._observer.is_alive():
            self._observer.stop()
            self._observer.join(timeout=3)
        self._pending.clear()

    def is_running(self) -> bool:
        return self._observer.is_alive()

    def _queue_event(
        self,
        event_type: str,
        src_path: str,
        dest_path: str = "",
    ) -> None:
        key = dest_path or src_path
        self._pending[key] = FileEvent(
            event_type=event_type,
            src_path=src_path,
            dest_path=dest_path,
            file_hash="",
            is_directory=False,
        )
        self._timer.start(self._debounce_ms)

    def _flush(self) -> None:
        if not self._pending:
            return
        batch = list(self._pending.values())
        self._pending.clear()
        self.file_system_changed.emit(batch)
```

```python
    def find_by_hash(self, file_hash: str) -> Optional[dict]:
        with self.connect() as conn:
            row = conn.execute(
                """
                SELECT id, file_name, full_path FROM files
                WHERE file_hash = ? AND is_deleted = 0
                """,
                (file_hash,),
            ).fetchone()
        return dict(row) if row else None

    def find_by_path_and_size(self, full_path: str, file_size: int) -> Optional[int]:
        with self.connect() as conn:
            row = conn.execute(
                """
                SELECT id FROM files
                WHERE full_path = ? AND file_size = ? AND is_deleted = 0
                LIMIT 1
                """,
                (full_path, file_size),
            ).fetchone()
        return row["id"] if row else None

    def update_file_path(self, file_id: int, new_path: str, new_name: str, new_ext: str) -> None:
        now = int(__import__("time").time())
        with self.connect() as conn:
            conn.execute(
                """
                UPDATE files
                SET full_path = ?, file_name = ?, file_ext = ?, updated_at = ?
                WHERE id = ?
                """,
                (new_path, new_name, new_ext, now, file_id),
            )
            conn.commit()

    def soft_delete_by_hash(self, file_hash: str) -> None:
        now = int(__import__("time").time())
        with self.connect() as conn:
            conn.execute(
                """
                UPDATE files
                SET is_deleted = 1, deleted_at = ?, updated_at = ?
                WHERE file_hash = ? AND is_deleted = 0
                """,
                (now, now, file_hash),
            )
            conn.commit()

    def update_file_meta(self, file_id: int, file_size: int, modify_time: int) -> None:
        now = int(__import__("time").time())
        with self.connect() as conn:
            conn.execute(
                """
                UPDATE files
                SET file_size = ?, modify_time = ?, updated_at = ?
                WHERE id = ?
                """,
                (file_size, modify_time, now, file_id),
            )
            conn.commit()

    def record_search(self, query_json: str, timestamp: int) -> None:
        with self.connect() as conn:
            conn.execute(
                "INSERT INTO search_history (query_json, created_at) VALUES (?, ?)",
                (query_json, timestamp),
            )
            conn.commit()

    def get_recent_searches(self, limit: int = 10) -> list[dict]:
        with self.connect() as conn:
            rows = conn.execute(
                """
                SELECT id, query_json, created_at
                FROM search_history
                ORDER BY created_at DESC
                LIMIT ?
                """,
                (limit,),
            ).fetchall()
        return [dict(r) for r in rows]

    def delete_old_searches(self, max_keep: int = 50) -> None:
        with self.connect() as conn:
            row = conn.execute(
                """
                SELECT id FROM search_history
                ORDER BY created_at DESC
                LIMIT 1 OFFSET ?
                """,
                (max_keep,),
            ).fetchone()
            if row:
                conn.execute(
                    "DELETE FROM search_history WHERE id <= ?",
                    (row["id"],),
                )
                conn.commit()
```

```python
    def _on_file_system_changed(self, events: list[FileEvent]) -> None:
        updated = False
        for ev in events:
            if ev.event_type == "modified":
                # Re-hash the file to get stable identifier
                p = Path(ev.src_path)
                if p.exists():
                    try:
                        st = p.stat()
                    except OSError:
                        continue
                    # Try matching by full_path + size first
                    file_id = self._db.find_by_path_and_size(ev.src_path, int(st.st_size))
                    if file_id is None:
                        # Fall back: compute hash
                        try:
                            file_hash = sha256_file(p)
                        except FileReadError:
                            continue
                        row = self._db.find_by_hash(file_hash)
                        if row:
                            file_id = row["id"]
                    if file_id is not None:
                        self._handle_file_modified(file_id, int(st.st_size), int(st.st_mtime))
                        updated = True
            elif ev.event_type == "deleted":
                self._handle_file_deleted_by_path(ev.src_path)
                updated = True
            elif ev.event_type == "moved":
                new_path = Path(ev.dest_path)
                if new_path.exists():
                    try:
                        file_hash = sha256_file(new_path)
                    except FileReadError:
                        continue
                    self._handle_file_moved(file_hash, ev.dest_path)
                    updated = True
            elif ev.event_type == "created":
                if self._auto_import:
                    p = Path(ev.src_path)
                    if p.exists():
                        try:
                            st = p.stat()
                        except OSError:
                            continue
                        cand = ImportCandidate(path=p, size=int(st.st_size))
                        self.import_files([cand])
                        updated = True
        if updated:
            self.signals.files_updated.emit([])

    def _handle_file_moved(self, file_hash: str, new_full_path: str) -> None:
        row = self._db.find_by_hash(file_hash)
        if row is None:
            return
        p = Path(new_full_path)
        self._db.update_file_path(
            file_id=row["id"],
            new_path=new_full_path,
            new_name=p.name,
            new_ext=p.suffix.lower().lstrip(".") if p.suffix else None,
        )

    def _handle_file_deleted(self, file_hash: str) -> None:
        self._db.soft_delete_by_hash(file_hash)

    def _handle_file_deleted_by_path(self, full_path: str) -> None:
        with self._db.connect() as conn:
            row = conn.execute(
                "SELECT id, file_size FROM files WHERE full_path = ? AND is_deleted = 0 LIMIT 1",
                (full_path,),
```

```python
from PyQt5.QtWidgets import QWidget, QVBoxLayout, QLineEdit, QListWidget, QListWidgetItem
from PyQt5.QtCore import pyqtSignal, Qt
from PyQt5.QtGui import QKeyEvent


class CommandPalette(QWidget):
    text_entered = pyqtSignal(str)
    tag_selected_for_file = pyqtSignal(int, int)

    def __init__(self, parent=None, vm=None):
        super().__init__(parent)
        self.vm = vm
        self.context_file_id = None
        self._history_items: list[str] = []
        self._history_index = -1

        layout = QVBoxLayout(self)
        self.search_box = QLineEdit()
        self.search_box.setPlaceholderText("搜索命令...")
        self.list_widget = QListWidget()
        layout.addWidget(self.search_box)
        layout.addWidget(self.list_widget)

        self.list_widget.itemActivated.connect(self._on_item_activated)
        self.search_box.textChanged.connect(self._on_text_changed)

    def set_context_file(self, file_id: int, file_name: str):
        self.context_file_id = file_id
        self._history_index = -1
        if file_id is not None:
            self.search_box.setPlaceholderText(f"为文件 {file_name} 添加标签...")
            self._load_tags()
        else:
            self.search_box.setPlaceholderText("搜索命令...")
            self.list_widget.clear()

    def set_history(self, items: list[dict]):
        self._history_items = []
        for item in items:
            import json
            try:
                data = json.loads(item["query_json"])
                q = data.get("query", "")
                if q:
                    self._history_items.append(q)
            except (json.JSONDecodeError, KeyError):
                continue
        self._history_index = -1
        if not self.context_file_id and self._history_items:
            self.list_widget.clear()
            for h in self._history_items:
                self.list_widget.addItem(QListWidgetItem(h))

    def keyPressEvent(self, event: QKeyEvent):
        if event.key() == Qt.Key_Up or event.key() == Qt.Key_Down:
            self._on_key_pressed(event.key())
            event.accept()
        else:
            super().keyPressEvent(event)

    def _on_key_pressed(self, key: int):
        if self.context_file_id is not None:
            return
        if not self._history_items:
            return
        n = len(self._history_items)
        if key == Qt.Key_Up:
            idx = max(0, self._history_index - 1) if self._history_index >= 0 else n - 1
        else:  # Key_Down
            if self._history_index >= n - 1:
                idx = -1
            else:
                idx = self._history_index + 1

        self._history_index = idx
        if 0 <= idx < n:
            self.search_box.setText(self._history_items[idx])
        else:
            self.search_box.clear()

    def _on_text_changed(self, text: str):
        if text:
            self._history_index = -1

    def _load_tags(self):
        self.list_widget.clear()
        if self.vm is not None:
            tags = self.vm.get_all_tags()
            for tag in tags:
                item = QListWidgetItem(tag.tag_name)
                item.setData(Qt.UserRole, tag.id)
                self.list_widget.addItem(item)

    def _on_item_activated(self, item):
        if self.context_file_id is not None:
            tag_id = item.data(Qt.UserRole)
            if tag_id is not None:
                self.tag_selected_for_file.emit(self.context_file_id, tag_id)
                self.hide()

    def hideEvent(self, event):
        super().hideEvent(event)
        self.set_context_file(None, "")
```
